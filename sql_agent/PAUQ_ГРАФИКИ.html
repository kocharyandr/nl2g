labels = data["labels"]

sorted_labels = [labels[i] for i in sorted_indices]

# Limit to top 10
top_10_labels = sorted_labels[:10]
top_10_values = sorted_values[:10]

bars = ax.bar(top_10_labels, top_10_values, color="steelblue", edgecolor="white", linewidth=0.8)

if data.get("title"): ax.set_title(data["title"], fontsize=14)

ax.tick_params(axis="x", rotation=45)

# Add value labels on top of bars
for b in bars:
    ax.text(b.get_x() + b.get_width()/2, b.get_height() + max(top_10_values)*0.01,
            f"{b.get_height():,.0f}", ha="center", va="bottom", fontsize=9)

fig.tight_layout()
